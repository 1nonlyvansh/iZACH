class NeuralCore(tk.Canvas):
    """Animated neural core — reacts when iZACH speaks."""

    def __init__(self, parent, size=200, **kw):
        super().__init__(parent, width=size, height=size,
                         bg=BG_DEEP, highlightthickness=0, **kw)
        self.size   = size
        self.cx     = size // 2
        self.cy     = size // 2
        self.speaking = False
        self._t     = 0
        self._nodes = self._gen_nodes()
        self._animate()

    def _gen_nodes(self):
        nodes = []
        rings = [(5, 55), (8, 90), (6, 35)]
        for count, radius in rings:
            for i in range(count):
                angle = (2 * math.pi / count) * i
                x = self.cx + radius * math.cos(angle)
                y = self.cy + radius * math.sin(angle)
                nodes.append({"x": x, "y": y, "base_r": radius,
                               "angle": angle, "phase": i * 0.4})
        return nodes

    def set_speaking(self, val: bool):
        self.speaking = val

    def _animate(self):
        self._t += 0.045
        t = self._t
        self.delete("all")

        pulse = 1.0 + (0.35 * math.sin(t * 3.5) if self.speaking else 0.08 * math.sin(t))

        # Outer glow ring
        glow_r = int(self.cx * 0.88 * pulse)
        for i, col in enumerate([CYAN_DARK, CYAN_DIM, "#004a55"]):
            r = glow_r + (3 - i) * 4
            self.create_oval(self.cx - r, self.cy - r,
                             self.cx + r, self.cy + r,
                             outline=col, width=1)

        # Draw edges between nearby nodes
        for i, a in enumerate(self._nodes):
            for j, b in enumerate(self._nodes):
                if j <= i:
                    continue
                dist = math.hypot(a["x"] - b["x"], a["y"] - b["y"])
                if dist < 70:
                    flicker = 0.3 + 0.5 * abs(math.sin(t + a["phase"] + b["phase"]))
                    if self.speaking:
                        flicker = min(1.0, flicker * 1.6)
                    col = self._lerp_color(CYAN_DARK, CYAN, flicker)
                    self.create_line(a["x"], a["y"], b["x"], b["y"],
                                     fill=col, width=1)

        # Draw nodes
        for nd in self._nodes:
            wave = math.sin(t * 1.8 + nd["phase"])
            r_mod = 1 + (0.25 * wave if self.speaking else 0.06 * wave)
            nr = 3.5 * r_mod
            glow = 0.4 + 0.6 * abs(wave) if self.speaking else 0.2 + 0.15 * abs(wave)
            col = self._lerp_color(CYAN_DIM, CYAN, glow)
            self.create_oval(nd["x"] - nr, nd["y"] - nr,
                             nd["x"] + nr, nd["y"] + nr,
                             fill=col, outline="")

        # Centre orb
        orb_r = int(14 * pulse)
        for layer in range(4, 0, -1):
            lr = orb_r + layer * 3
            alpha = ["0a", "14", "22", "33"][layer - 1]
            self.create_oval(self.cx - lr, self.cy - lr,
                             self.cx + lr, self.cy + lr,
                             fill="", outline=CYAN, width=1)
        self.create_oval(self.cx - orb_r, self.cy - orb_r,
                         self.cx + orb_r, self.cy + orb_r,
                         fill=CYAN if self.speaking else CYAN_DIM, outline="")

        self.after(33, self._animate)  # ~30 fps

    @staticmethod
    def _lerp_color(c1, c2, t):
        t = max(0.0, min(1.0, t))
        r1, g1, b1 = int(c1[1:3], 16), int(c1[3:5], 16), int(c1[5:7], 16)
        r2, g2, b2 = int(c2[1:3], 16), int(c2[3:5], 16), int(c2[5:7], 16)
        r = int(r1 + (r2 - r1) * t)
        g = int(g1 + (g2 - g1) * t)
        b = int(b1 + (b2 - b1) * t)
        return f"#{r:02x}{g:02x}{b:02x}"

class ChatPanel(tk.Frame):
    def __init__(self, parent, **kw):
        super().__init__(parent, bg=BG_CARD, **kw)
        self._build()

    def _build(self):
        self.canvas = tk.Canvas(self, bg=BG_CARD, highlightthickness=0)
        self.scroll = tk.Scrollbar(self, orient="vertical",
                                   command=self.canvas.yview,
                                   bg=BG_PANEL, troughcolor=BG_DEEP,
                                   activebackground=CYAN)
        self.canvas.configure(yscrollcommand=self.scroll.set)
        self.scroll.pack(side="right", fill="y")
        self.canvas.pack(side="left", fill="both", expand=True)

        self.inner = tk.Frame(self.canvas, bg=BG_CARD)
        self._win = self.canvas.create_window((0, 0), window=self.inner, anchor="nw")
        self.inner.bind("<Configure>", self._on_configure)
        self.canvas.bind("<Configure>", self._on_canvas_resize)

    def _on_configure(self, _e):
        self.canvas.configure(scrollregion=self.canvas.bbox("all"))
        self.canvas.yview_moveto(1.0)

    def _on_canvas_resize(self, e):
        self.canvas.itemconfig(self._win, width=e.width)

    def add_message(self, sender: str, text: str):
        is_user = sender.upper() == "USER"
        row = tk.Frame(self.inner, bg=BG_CARD)
        row.pack(fill="x", padx=10, pady=4)

        bubble_bg = "#0d2535" if not is_user else "#0a2a15"
        bubble_fg = CYAN if not is_user else GREEN
        anchor_side = "w" if not is_user else "e"

        outer = tk.Frame(row, bg=BG_CARD)
        outer.pack(anchor=anchor_side)

        name_lbl = tk.Label(outer,
                             text="iZACH" if not is_user else "YOU",
                             bg=BG_CARD,
                             fg=bubble_fg,
                             font=("Consolas", 8, "bold"))
        name_lbl.pack(anchor=anchor_side, padx=4)

        bubble = tk.Frame(outer,
                          bg=bubble_bg,
                          highlightthickness=1,
                          highlightbackground=bubble_fg if not is_user else GREEN_DIM)
        bubble.pack(anchor=anchor_side)

        msg = tk.Label(bubble,
                       text=text,
                       bg=bubble_bg,
                       fg=TEXT_PRI,
                       font=("Consolas", 10),
                       wraplength=300,
                       justify="left" if not is_user else "right",
                       padx=10, pady=6)
        msg.pack()

class StatsPanel(tk.Frame):
    def __init__(self, parent, **kw):
        super().__init__(parent, bg=BG_CARD, **kw)
        self._build()
        self.after(100, self._update)

    def _build(self):
        header = tk.Label(self, text="* SYSTEM VITALS",
                          bg=BG_CARD, fg=CYAN,
                          font=("Consolas", 9, "bold"))
        header.pack(anchor="w", padx=10, pady=(8, 4))

        self._bars = {}
        for label in ["CPU", "RAM", "GPU"]:
            row = tk.Frame(self, bg=BG_CARD)
            row.pack(fill="x", padx=10, pady=2)
            lbl = tk.Label(row, text=f"{label:<4}", bg=BG_CARD, fg=TEXT_SEC,
                           font=("Consolas", 9), width=4)
            lbl.pack(side="left")
            bar_bg = tk.Frame(row, bg=BORDER, height=6)
            bar_bg.pack(side="left", fill="x", expand=True, padx=(4, 8))
            bar_fill = tk.Frame(bar_bg, bg=CYAN, height=6)
            bar_fill.place(x=0, y=0, relheight=1.0, relwidth=0.0)
            val_lbl = tk.Label(row, text="0%", bg=BG_CARD, fg=CYAN,
                               font=("Consolas", 9), width=5)
            val_lbl.pack(side="right")
            self._bars[label] = {"bg": bar_bg, "fill": bar_fill, "val": val_lbl}

        sep = tk.Frame(self, bg=BORDER, height=1)
        sep.pack(fill="x", padx=10, pady=4)

        izach_lbl = tk.Label(self, text="* iZACH PROCESS",
                             bg=BG_CARD, fg=AMBER,
                             font=("Consolas", 9, "bold"))
        izach_lbl.pack(anchor="w", padx=10, pady=(0, 4))

        self._proc_bars = {}
        for label in ["CPU", "MEM"]:
            row = tk.Frame(self, bg=BG_CARD)
            row.pack(fill="x", padx=10, pady=2)
            lbl = tk.Label(row, text=f"{label:<4}", bg=BG_CARD, fg=TEXT_SEC,
                           font=("Consolas", 9), width=4)
            lbl.pack(side="left")
            bar_bg = tk.Frame(row, bg=BORDER, height=6)
            bar_bg.pack(side="left", fill="x", expand=True, padx=(4, 8))
            bar_fill = tk.Frame(bar_bg, bg=AMBER, height=6)
            bar_fill.place(x=0, y=0, relheight=1.0, relwidth=0.0)
            val_lbl = tk.Label(row, text="0%", bg=BG_CARD, fg=AMBER,
                               font=("Consolas", 9), width=5)
            val_lbl.pack(side="right")
            self._proc_bars[label] = {"bg": bar_bg, "fill": bar_fill, "val": val_lbl}

    def _update(self):
        cpu = psutil.cpu_percent()
        ram = psutil.virtual_memory().percent
        gpu = 0
        try:
            out = subprocess.check_output(
                ["nvidia-smi", "--query-gpu=utilization.gpu",
                 "--format=csv,noheader,nounits"],
                stderr=subprocess.DEVNULL).decode().strip()
            gpu = float(out.split("\n")[0])
        except Exception:
            pass

        for label, val in [("CPU", cpu), ("RAM", ram), ("GPU", gpu)]:
            b = self._bars[label]
            b["fill"].place(relwidth=val / 100)
            b["val"].config(text=f"{val:.0f}%")
            color = RED if val > 85 else (AMBER if val > 65 else CYAN)
            b["fill"].config(bg=color)

        # iZACH process stats
        try:
            proc = psutil.Process(os.getpid())
            p_cpu = proc.cpu_percent(interval=None)
            p_mem = proc.memory_percent()
        except Exception:
            p_cpu, p_mem = 0, 0

        for label, val in [("CPU", p_cpu), ("MEM", p_mem)]:
            b = self._proc_bars[label]
            b["fill"].place(relwidth=min(val / 100, 1.0))
            b["val"].config(text=f"{val:.1f}%")

        self.after(1500, self._update)

class SpotifyPanel(tk.Frame):
    def __init__(self, parent, spotify_handler=None, **kw):
        super().__init__(parent, bg=BG_CARD, **kw)
        self.sp = spotify_handler
        self._build()
        self._poll()

    def _build(self):
        header = tk.Label(self, text="* SPOTIFY",
                          bg=BG_CARD, fg=GREEN,
                          font=("Consolas", 9, "bold"))
        header.pack(anchor="w", padx=10, pady=(8, 6))

        center = tk.Frame(self, bg=BG_CARD)
        center.pack(fill="x", padx=10)

        self._art_label = tk.Label(center, bg=BG_CARD)
        self._art_label.pack(side="left", padx=(0, 10))
        self._set_default_art()

        info = tk.Frame(center, bg=BG_CARD)
        info.pack(side="left", fill="both", expand=True)

        self._track_var = tk.StringVar(value="─ Nothing playing ─")
        self._artist_var = tk.StringVar(value="")

        tk.Label(info, textvariable=self._track_var,
                 bg=BG_CARD, fg=TEXT_PRI,
                 font=("Consolas", 10, "bold"),
                 wraplength=160, justify="left").pack(anchor="w")

        tk.Label(info, textvariable=self._artist_var,
                 bg=BG_CARD, fg=GREEN,
                 font=("Consolas", 9)).pack(anchor="w")

        # Controls row
        ctrl = tk.Frame(self, bg=BG_CARD)
        ctrl.pack(fill="x", padx=10, pady=6)

        btn_cfg = {"bg": BG_PANEL, "fg": CYAN, "font": ("Consolas", 10),
                   "relief": "flat", "cursor": "hand2",
                   "activebackground": CYAN_DARK, "activeforeground": CYAN,
                   "padx": 8, "pady": 2}

        tk.Button(ctrl, text="◀◀", command=self._prev, **btn_cfg).pack(side="left", padx=2)
        tk.Button(ctrl, text="▶ ❚❚", command=self._play_pause, **btn_cfg).pack(side="left", padx=2)
        tk.Button(ctrl, text="▶▶", command=self._next, **btn_cfg).pack(side="left", padx=2)

        # Device switcher
        sep = tk.Frame(self, bg=BORDER, height=1)
        sep.pack(fill="x", padx=10, pady=(2, 4))

        dev_row = tk.Frame(self, bg=BG_CARD)
        dev_row.pack(fill="x", padx=10, pady=(0, 6))

        tk.Label(dev_row, text="DEVICE", bg=BG_CARD, fg=TEXT_SEC,
                 font=("Consolas", 8)).pack(side="left")

        self._device_var = tk.StringVar(value="─")
        self._device_menu = tk.OptionMenu(dev_row, self._device_var, "─",
                                          command=self._switch_device)
        self._device_menu.config(bg=BG_PANEL, fg=CYAN,
                                 font=("Consolas", 9),
                                 relief="flat", borderwidth=0,
                                 highlightthickness=1,
                                 highlightbackground=BORDER,
                                 activebackground=CYAN_DARK,
                                 activeforeground=CYAN)
        self._device_menu["menu"].config(bg=BG_PANEL, fg=CYAN,
                                         font=("Consolas", 9))
        self._device_menu.pack(side="left", padx=6)

        tk.Button(dev_row, text="↺", font=("Consolas", 10),
                  bg=BG_PANEL, fg=TEXT_SEC, relief="flat",
                  command=self._refresh_devices,
                  cursor="hand2").pack(side="left")

    def _set_default_art(self):
        img = Image.new("RGB", (52, 52), color="#1a1a2e")
        draw = ImageDraw.Draw(img)
        draw.ellipse([14, 14, 38, 38], fill=GREEN_DIM, outline=GREEN)
        draw.ellipse([22, 22, 30, 30], fill="#1a1a2e")
        self._art_photo = ImageTk.PhotoImage(img)
        self._art_label.config(image=self._art_photo)

    def _poll(self):
        if self.sp:
            try:
                current = self.sp.sp.current_playback()
                if current and current.get("item"):
                    item = current["item"]
                    track = item["name"]
                    artist = ", ".join(a["name"] for a in item["artists"])
                    self._track_var.set(track[:28] + "…" if len(track) > 28 else track)
                    self._artist_var.set(artist[:28] + "…" if len(artist) > 28 else artist)

                    # Album art
                    try:
                        images = item.get("album", {}).get("images", [])
                        if images:
                            url = images[-1]["url"]
                            resp = requests.get(url, timeout=3)
                            img = Image.open(__import__("io").BytesIO(resp.content))
                            img = img.resize((52, 52), Image.Resampling.LANCZOS)
                            self._art_photo = ImageTk.PhotoImage(img)
                            self._art_label.config(image=self._art_photo)
                    except Exception:
                        pass
                else:
                    self._track_var.set("─ Nothing playing ─")
                    self._artist_var.set("")
                    self._set_default_art()
            except Exception:
                pass
        self.after(5000, self._poll)

    def _refresh_devices(self):
        if not self.sp:
            return
        try:
            devices = self.sp.sp.devices().get("devices", [])
            menu = self._device_menu["menu"]
            menu.delete(0, "end")
            for d in devices:
                name = d["name"]
                menu.add_command(label=name,
                                 command=lambda n=name: self._device_var.set(n))
            if devices:
                self._device_var.set(devices[0]["name"])
        except Exception:
            pass

    def _switch_device(self, name):
        if self.sp and name != "─":
            threading.Thread(target=self.sp.switch_device,
                             args=(name,), daemon=True).start()

    def _prev(self):
        if self.sp:
            threading.Thread(target=self.sp.previous_track, daemon=True).start()

    def _next(self):
        if self.sp:
            threading.Thread(target=self.sp.next_track, daemon=True).start()

    def _play_pause(self):
        if not self.sp:
            return
        try:
            current = self.sp.sp.current_playback()
            if current and current.get("is_playing"):
                threading.Thread(target=self.sp.pause_music, daemon=True).start()
            else:
                threading.Thread(target=self.sp.resume_music, daemon=True).start()
        except Exception:
            pass

class CameraPanel(tk.Frame):
    def __init__(self, parent):
        super().__init__(parent, bg=BG_CARD)
        self._cap = None
        self._running = False
        self.width = 240
        self.height = 180
        self._build()
        

    def _build(self):
        header = tk.Label(self, text="* CAMERA FEED",
                          bg=BG_CARD, fg=CYAN,
                          font=("Consolas", 9, "bold"))
        header.pack(anchor="w", padx=10, pady=(8, 4))

        self._canvas = tk.Canvas(self, bg="#000000",
                                 width=self.width, height=self.height,
                                 highlightthickness=1,
                                 highlightbackground=BORDER)
        self._canvas.pack(padx=10, pady=(0, 4))

        self._status_lbl = tk.Label(self, text="> CAMERA OFFLINE",
                                    bg=BG_CARD, fg=RED,
                                    font=("Consolas", 8))
        self._status_lbl.pack(anchor="w", padx=10, pady=(0, 6))

        btn_row = tk.Frame(self, bg=BG_CARD)
        btn_row.pack(fill="x", padx=10, pady=(0, 6))

        tk.Button(btn_row, text="START CAM",
                  bg=GREEN_DIM, fg=GREEN,
                  font=("Consolas", 9, "bold"),
                  relief="flat", cursor="hand2",
                  activebackground="#0a3a15",
                  command=self._start).pack(side="left", padx=(0, 6))

        tk.Button(btn_row, text="STOP",
                  bg="#2a0000", fg=RED,
                  font=("Consolas", 9, "bold"),
                  relief="flat", cursor="hand2",
                  activebackground="#3a0000",
                  command=self._stop).pack(side="left")

        self._draw_offline()

    def _draw_offline(self):
        self._canvas.delete("all")
        self._canvas.create_rectangle(0, 0, self.width, self.height, fill="#050505")
        self._canvas.create_text(self.width // 2, self.height // 2, text="[ NO SIGNAL ]",
                                 fill=TEXT_SEC, font=("Consolas", 10))

    def _start(self):
        if self._running:
            return
        self._cap = cv2.VideoCapture(0)
        if not self._cap.isOpened():
            self._status_lbl.config(text="> CAMERA NOT FOUND", fg=RED)
            return
        self._running = True
        self._status_lbl.config(text="> CAMERA LIVE", fg=GREEN)
        threading.Thread(target=self._feed_loop, daemon=True).start()

    def _stop(self):
        self._running = False
        if self._cap:
            self._cap.release()
            self._cap = None
        self._draw_offline()
        self._status_lbl.config(text="> CAMERA OFFLINE", fg=RED)

    def _feed_loop(self):
        while self._running and self._cap:
            ret, frame = self._cap.read()
            if not ret:
                break
            h, w, _ = frame.shape

            target_w = self.width
            target_h = self.height

            scale = min(target_w / w, target_h / h)

            new_w = int(w * scale)
            new_h = int(h * scale)

            frame = cv2.resize(frame, (new_w, new_h))
            frame = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
            img = Image.fromarray(frame)
            photo = ImageTk.PhotoImage(img)
            try:
                x_offset = (self.width - new_w) // 2
                y_offset = (self.height - new_h) // 2

                self._canvas.create_image(x_offset, y_offset, anchor="nw", image=photo)
                self._canvas._photo = photo  # keep reference
            except Exception:
                break
            time.sleep(0.033)

    # ─────────────────────────────────────────
    # PUBLIC API (called by main.py)
    # ─────────────────────────────────────────
    def write_log(self, text: str):
        if ":" in text:
            sender, message = text.split(":", 1)
            self._add_message(sender.strip(), message.strip())
        else:
            self._add_message("SYSTEM", text)

    def _add_message(self, sender: str, message: str):
        self.root.after(0, lambda: self._chat.add_message(sender, message))

    def update_status(self, text: str, is_warning=False):
        color = AMBER if is_warning else GREEN
        self.root.after(0, lambda: (
            self._status_var.set(f"> {text.upper()}"),
        ))

    def set_speaking(self, val: bool):
        """Call this when iZACH starts/stops speaking so the orb reacts."""
        self.root.after(0, lambda: self._neural.set_speaking(val))

    def is_mic_active(self) -> bool:
        return self._mic_active

    # ─────────────────────────────────────────
    # INTERNAL
    # ─────────────────────────────────────────
    def _toggle_mic(self):
        self._mic_active = not self._mic_active
        if self._mic_active:
            self._mic_btn.config(text="🎤  MIC ON",
                                 bg=GREEN_DIM, fg=GREEN)
        else:
            self._mic_btn.config(text="🔇  MIC OFF",
                                 bg="#2a0000", fg=RED)

    def _send(self):
        text = self._entry.get().strip()
        if not text:
            return
        self._entry.delete(0, "end")
        self._add_message("USER", text)
        if self._chain:
            threading.Thread(target=self._chain.process,
                             args=(text,), daemon=True).start()

    def _update_ticker(self):
        now = time.strftime("%Y-%m-%d  %H:%M:%S")
        cpu = psutil.cpu_percent()
        ram = psutil.virtual_memory().percent
        self._ticker_var.set(
            f"  {now}     CPU {cpu:.0f}%     RAM {ram:.0f}%     "
            f"iZACH v9.0  —  NEURAL LINK ACTIVE"
        )
        self.root.after(1000, self._update_ticker)

    def run(self):
        self.root.mainloop()
